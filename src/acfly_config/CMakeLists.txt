cmake_minimum_required(VERSION 3.8)
project(acfly_config)
find_package(ament_cmake REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl REQUIRED)
find_package(rcl_yaml_param_parser REQUIRED)
install(DIRECTORY include/ DESTINATION include)
ament_export_include_directories(include)
ament_export_dependencies(ament_index_cpp rclcpp rcl rcl_yaml_param_parser)
if(BUILD_TESTING)
  enable_testing()
  add_executable(config_probe test/config_probe.cpp)
  target_include_directories(config_probe PRIVATE include)
  target_compile_features(config_probe PRIVATE cxx_std_17)
  ament_target_dependencies(config_probe ament_index_cpp rclcpp rcl rcl_yaml_param_parser)
  install(DIRECTORY test/fixtures/ DESTINATION share/${PROJECT_NAME}/config)
  add_test(NAME default_category_files COMMAND config_probe 7)
  add_test(NAME operator_precedence COMMAND config_probe 19 --ros-args
    --params-file ${CMAKE_CURRENT_SOURCE_DIR}/test/fixtures/override.yaml -p sample_rate:=19)
  add_test(NAME operator_file_precedence COMMAND config_probe 11 --ros-args
    --params-file ${CMAKE_CURRENT_SOURCE_DIR}/test/fixtures/override.yaml)
  add_test(NAME category_required COMMAND config_probe missing)
  add_test(NAME refuse_wrong_type COMMAND config_probe wrong_type --ros-args -p sample_rate:=oops)
  add_test(NAME refuse_unknown_key COMMAND config_probe unknown --ros-args -p typo:=7)
  add_test(NAME refuse_conflicting_overrides COMMAND config_probe wrong_type --ros-args
    --params-file ${CMAKE_CURRENT_SOURCE_DIR}/test/fixtures/override.yaml -p sample_rate:=oops)
  set_tests_properties(refuse_conflicting_overrides refuse_wrong_type refuse_unknown_key default_category_files operator_precedence operator_file_precedence category_required
    PROPERTIES ENVIRONMENT "ROS_LOCALHOST_ONLY=1;ROS_DOMAIN_ID=188")
endif()
ament_package()
